cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(DimensionAgentNative VERSION 0.7.28 LANGUAGES CXX)
# Some localized MSVC installations report a UTF-8 include prefix that
# CMake decodes using the console code page during compiler detection.
if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja" AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^娉")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX_CXX "${CMAKE_CL_SHOWINCLUDES_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(dimension_core STATIC src/json.cpp src/scheduler.cpp src/store.cpp src/archive.cpp src/archive_export.cpp src/archive_staging.cpp src/model.cpp src/http.cpp src/runs.cpp src/process.cpp src/process_session.cpp src/desktop.cpp src/desktop_homes.cpp)
target_sources(dimension_core PRIVATE src/application_bridge.cpp)
target_sources(dimension_core PRIVATE src/image_assets.cpp)
target_sources(dimension_core PRIVATE src/approvals.cpp src/approval_tools.cpp)
target_sources(dimension_core PRIVATE src/message_context.cpp src/chat_work_bridge.cpp)
target_sources(dimension_core PRIVATE src/workspaces.cpp)
target_sources(dimension_core PRIVATE src/model_catalog.cpp src/work_benchmark.cpp src/model_routing.cpp)
target_sources(dimension_core PRIVATE src/atomic_file.cpp src/notification_inbox.cpp src/character_tools.cpp src/preview.cpp src/sandbox_exec.cpp)
target_sources(dimension_core PRIVATE src/conversation_services.cpp src/persona_fields.cpp)
target_sources(dimension_core PRIVATE src/work_compaction.cpp src/proactive_reminders.cpp src/work_templates.cpp)
target_sources(dimension_core PRIVATE src/bot_archive.cpp src/task_handoff.cpp)
target_sources(dimension_core PRIVATE src/file_ops.cpp src/host_exec.cpp src/hybrid_workspace.cpp src/cli_stream_driver.cpp src/bot_capabilities.cpp src/chat_realism.cpp src/chat_knowledge.cpp src/breeze_tts.cpp src/living_memory.cpp src/network_proxy.cpp src/work_cli_broker.cpp src/cli_context_bus.cpp src/work_craft_evolution.cpp src/work_checkpoints.cpp src/canvas_engine.cpp)
target_include_directories(dimension_core PUBLIC include)
target_link_libraries(dimension_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(dimension_core PUBLIC winhttp ws2_32 bcrypt ole32 shell32 windowscodecs crypt32 advapi32)
endif()
if(MSVC)
  target_compile_options(dimension_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
  add_link_options(/HIGHENTROPYVA /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
endif()
add_executable(DImensionagent-core src/main.cpp)
target_link_libraries(DImensionagent-core PRIVATE dimension_core)
add_executable(DImensionagent-import-bots src/bot_memory_import_main.cpp)
target_link_libraries(DImensionagent-import-bots PRIVATE dimension_core)
add_executable(DImensionagent-native src/server_main.cpp src/app.cpp)
add_executable(sandbox_exec_test tests/sandbox_exec_test.cpp)
target_link_libraries(sandbox_exec_test PRIVATE dimension_core)
target_link_libraries(DImensionagent-native PRIVATE dimension_core crypt32)
if(WIN32)
  enable_language(RC)
  set(WEBVIEW2_SDK "${CMAKE_SOURCE_DIR}/third_party/webview2/1.0.4314.35" CACHE PATH "Microsoft WebView2 SDK")
  add_executable(DImensionagent WIN32 src/desktop_host.cpp src/json.cpp src/desktop_host.rc src/desktop_host.manifest)
  target_include_directories(DImensionagent PRIVATE include "${WEBVIEW2_SDK}/build/native/include")
  target_link_libraries(DImensionagent PRIVATE user32 shell32 ole32 gdiplus dwmapi)
  if(MSVC)
    target_compile_options(DImensionagent PRIVATE /utf-8 /W4 /permissive- /EHsc)
  endif()
endif()
enable_testing()
foreach(extra IN ITEMS image_assets memory_access group_status approvals workspaces message_context chat_work_bridge mid_memory archive_mid model_policy atomic_file notification_inbox character_tools preview conversation_services brain_space episode_memory impression_memory bot_archive task_handoff work_compaction persona_fields proactive_reminders file_ops hybrid_workspace studio_backend chat_realism chat_knowledge breeze_tts living_memory network_proxy work_cli_broker cli_context_bus work_craft_evolution work_checkpoints canvas_engine)
  add_executable(${extra}_test tests/${extra}_test.cpp)
  target_link_libraries(${extra}_test PRIVATE dimension_core)
  target_compile_definitions(${extra}_test PRIVATE DIMENSION_TEST_ROOT=L"${CMAKE_BINARY_DIR}/test-data/${extra}")
  add_test(NAME ${extra} COMMAND ${extra}_test ${CMAKE_BINARY_DIR}/test-data/${extra})
  set_tests_properties(${extra} PROPERTIES TIMEOUT 45)
  if(extra STREQUAL "model_policy" OR extra STREQUAL "approvals")
    set_tests_properties(${extra} PROPERTIES TIMEOUT 120)
  endif()
endforeach()
set_tests_properties(conversation_services PROPERTIES TIMEOUT 90)
add_executable(application_bridge_test tests/application_bridge_test.cpp)
target_link_libraries(application_bridge_test PRIVATE dimension_core)
add_test(NAME application_bridge COMMAND application_bridge_test ${CMAKE_BINARY_DIR}/test-data/application_bridge)
set_tests_properties(application_bridge PROPERTIES TIMEOUT 45)
foreach(module IN ITEMS json scheduler store store_import store_groups store_snapshot archive archive_groups archive_export archive_staging relation_notes bot_rosters model http runs process process_session desktop desktop_homes)
  add_executable(${module}_test tests/${module}_test.cpp)
  target_link_libraries(${module}_test PRIVATE dimension_core)
  if(module STREQUAL "model" OR module STREQUAL "http")
    add_test(NAME ${module} COMMAND ${module}_test)
  else()
    add_test(NAME ${module} COMMAND ${module}_test ${CMAKE_BINARY_DIR}/test-data/${module})
  endif()
  set_tests_properties(${module} PROPERTIES TIMEOUT 45)
  if(module STREQUAL "process_session")
    set_tests_properties(${module} PROPERTIES TIMEOUT 120)
  endif()
endforeach()
